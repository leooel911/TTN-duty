def parse_transport_periods(raw_periods, year=2026):
    expanded = {}
    for k, v in raw_periods.items():
        if "-" in k:
            parts = k.split("-")
            s_m, s_d = map(int, parts[0].strip().split("/"))
            e_m, e_d = map(int, parts[1].strip().split("/"))
            cur = date(year, s_m, s_d)
            end_dt = date(year, e_m, e_d)
            while cur <= end_dt:
                expanded[f"{cur.month}/{cur.day}"] = v
                cur += timedelta(days=1)
        else:
            expanded[k.strip()] = v
    return expanded

def pad_time(t_str):
    if not t_str or ":" not in t_str:
        return t_str
    parts = t_str.split(":")
    if len(parts) == 2:
        return f"{int(parts[0]):02d}:{parts[1]}"
    return t_str

def fmt_hours(h):
    if not h or "h" in h:
        return h
    if ":" in h:
        p = h.split(":")
        return f"{p[0]}h{p[1]}m"
    return h

def parse_cell(raw):
    lines = [l.strip() for l in raw.strip().split("\n") if l.strip()]
    if not lines:
        return dict(start="", train="", end="", hours="", note="")
    if len(lines) == 1 and ("DO" in lines[0] or "D2W" in lines[0]):
        return dict(start="", train=lines[0], end="", hours="", note="")
    if "PAY" in lines:
        time_pattern = re.compile(r'^\d{1,2}:\d{2}$')
        times = [l for l in lines if time_pattern.match(l)]
        start_t = pad_time(times[0]) if len(times) >= 1 else ""
        end_t = pad_time(times[1]) if len(times) >= 2 else ""
        hours = ""
        for l in reversed(lines):
            if "h" in l or (":" in l and l not in times):
                hours = fmt_hours(l)
                break
        return dict(start=start_t, train="PAY", end=end_t, hours=hours, note="")

    time_match_pattern = re.compile(r'^(\d{1,2}:\d{2})$')
    times = []
    other_lines = []
    hours = ""
    for l in lines:
        if time_match_pattern.match(l):
            times.append(l)
        elif "h" in l or "m" in l:
            hours = fmt_hours(l)
        else:
            other_lines.append(l)

    if len(times) >= 3 and not hours:
        hours = fmt_hours(times.pop())

    start_time = pad_time(times[0]) if len(times) >= 1 else ""
    end_time = pad_time(times[1]) if len(times) >= 2 else ""

    train_code = ""
    notes = []
    for item in other_lines:
        if item.startswith("DO") or item.startswith("D2W"):
            notes.append(item)
        elif not train_code:
            train_code = item
        else:
            notes.append(item)

    return dict(start=start_time, end=end_time, train=train_code, hours=hours, note=" ".join(notes))

def parse_flexible_employees(text, target_name):
    f = io.StringIO(text.strip())
    reader = csv.reader(f, delimiter="\t")
    rows = list(reader)

    start_date_str = None
    for r in rows:
        if not r or len(r) < 3:
            continue
        first_cell = r[0].strip()
        if any(k in first_cell for k in ["員工", "編號", "工號", "Emp", "No"]):
            possible_date = r[2].strip()
            if "/" in possible_date:
                match_date = re.search(r'(\d+/\d+)', possible_date)
                if match_date:
                    start_date_str = match_date.group(1)
                    break

    if not start_date_str:
        raise ValueError("文字檔中找不到包含日期的標題列，請檢查複製內容。")

    year = 2026
    start_m, start_d = map(int, start_date_str.split("/"))
    start_dt = date(year, start_m, start_d)

    employees = []
    for r in rows:
        if not r or len(r) < 2:
            continue
        if any(k in r[0].strip() for k in ["員工", "編號", "工號"]):
            continue
        match = re.search(r'(A\d{6})', r[0].strip())
        if match:
            emp_id = match.group(1)
            emp_name = r[1].strip()
            if emp_name == target_name.strip():
                cells = r[2:]
                employees.append((emp_id, emp_name, cells))

    if not employees:
        raise ValueError(f"找不到同仁「{target_name}」的資料，請確認名字是否正確。")

    base_days = len(employees[0][2])
    dates = [f"{(start_dt + timedelta(days=i)).month}/{(start_dt + timedelta(days=i)).day}" for i in range(base_days)]
    return start_dt, dates, employees[0], start_date_str

def is_overtime(h):
    if not h:
        return False
    try:
        p = h.replace("h", ":").replace("m", "").split(":")
        return (int(p[0]) * 60 + int(p[1])) > 510
    except Exception:
        return False

def is_town_shift(tr, note):
    target_keywords = ["TOWN", "STD", "TTN", "DTT", "工廠", "回廠", "訓練"]
    combined = f"{tr} {note}".upper()
    return any(kw in combined for kw in target_keywords)

def build_weeks(start_dt, dates, cells):
    start_weekday = start_dt.weekday()
    first_wd = (start_weekday + 1) % 7
    weeks, week = [], [None] * first_wd
    for dt, raw in zip(dates, cells):
        week.append((dt, parse_cell(raw)))
        if len(week) == 7:
            weeks.append(week)
            week = []
    if week:
        while len(week) < 7:
            week.append(None)
        weeks.append(week)
    return weeks

def setup_font():
    font_paths = [
        "/System/Library/Fonts/PingFang.ttc",
        "/usr/share/fonts/opentype/noto/NotoSansCJK-Regular.ttc",
        "C:\\Windows\\Fonts\\msjh.ttc",
    ]
    for path in font_paths:
        if os.path.exists(path):
            try:
                fm.fontManager.addfont(path)
                return fm.FontProperties(fname=path)
            except Exception:
                continue
    return None
